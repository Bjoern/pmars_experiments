cmake_minimum_required(VERSION 3.21)
project(pmars_sdl3 VERSION 0.9.6 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
if(WIN32 AND (NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8))
  message(FATAL_ERROR "Windows builds require 64-bit MinGW-w64 GCC. Set CMAKE_C_COMPILER and CMAKE_CXX_COMPILER explicitly in a fresh build directory.")
endif()
option(PMARS_STATIC "Link SDL3 and MinGW compiler runtimes statically" ON)
set(SDL3_SOURCE_DIR "" CACHE PATH "Optional SDL3 source directory")
if(PMARS_STATIC)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
# The source bundle carries SDL beside pMARS; also accept dependencies inside
# the pMARS tree. No implicit downloads or machine-specific paths are needed.
if(NOT SDL3_SOURCE_DIR)
  foreach(deps "${CMAKE_CURRENT_SOURCE_DIR}/dependencies" "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies")
    if(EXISTS "${deps}/SDL3-3.4.16/CMakeLists.txt")
      set(SDL3_SOURCE_DIR "${deps}/SDL3-3.4.16")
      break()
    elseif(EXISTS "${deps}/SDL3-3.4.16.zip")
      file(SHA256 "${deps}/SDL3-3.4.16.zip" SDL3_ARCHIVE_HASH)
      if(NOT SDL3_ARCHIVE_HASH STREQUAL "5f399fdfbc040169ef4418a3cb99ccc5a0641681f7ccc1da66eeec3c97fa4076")
        message(FATAL_ERROR "SDL3 source archive SHA-256 mismatch: ${deps}/SDL3-3.4.16.zip")
      endif()
      set(SDL3_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/_deps/SDL3-3.4.16")
      if(NOT EXISTS "${SDL3_SOURCE_DIR}/CMakeLists.txt")
        file(ARCHIVE_EXTRACT INPUT "${deps}/SDL3-3.4.16.zip"
             DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/_deps")
      endif()
      break()
    endif()
  endforeach()
endif()
if(SDL3_SOURCE_DIR)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SDL3_SOURCE_DIR}" sdl3 EXCLUDE_FROM_ALL)
else()
  if(PMARS_STATIC)
    find_package(SDL3 3.2 REQUIRED CONFIG COMPONENTS SDL3-static)
  else()
    find_package(SDL3 3.2 REQUIRED CONFIG)
  endif()
endif()
set(PMARS_SOURCES src/pmars.c src/asm.c src/eval.c src/disasm.c src/cdb.c
    src/sim.c src/pos.c src/clparse.c src/global.c src/token.c src/str_eng.c)
add_executable(pmars-sdl3 ${PMARS_SOURCES} src/SDL_main.c)
target_compile_definitions(pmars-sdl3 PRIVATE SDLGRAPHX EXT94 PERMUTATE)
if(PMARS_STATIC)
  target_link_libraries(pmars-sdl3 PRIVATE SDL3::SDL3-static)
else()
  target_link_libraries(pmars-sdl3 PRIVATE SDL3::SDL3)
endif()
if(MINGW)
  target_link_options(pmars-sdl3 PRIVATE -mconsole -Wl,--dynamicbase,--nxcompat)
  if(PMARS_STATIC)
    target_link_options(pmars-sdl3 PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pmars-sdl3 PRIVATE -fstack-protector-strong)
endif()
add_executable(pmars-server ${PMARS_SOURCES})
target_compile_definitions(pmars-server PRIVATE SERVER EXT94 PERMUTATE)
if(MINGW AND PMARS_STATIC)
  target_link_options(pmars-server PRIVATE -static -static-libgcc)
endif()

option(PMARS_BUILD_EVENT_TESTS "Build SDL3 event/renderer integration executable" OFF)
if(PMARS_BUILD_EVENT_TESTS)
  add_executable(pmars-sdl3-events ${PMARS_SOURCES} src/SDL_main.c)
  target_compile_definitions(pmars-sdl3-events PRIVATE SDLGRAPHX EXT94 PERMUTATE PMARS_SDL3_TEST)
  if(PMARS_STATIC)
    target_link_libraries(pmars-sdl3-events PRIVATE SDL3::SDL3-static)
  else()
    target_link_libraries(pmars-sdl3-events PRIVATE SDL3::SDL3)
  endif()
  if(MINGW AND PMARS_STATIC)
    target_link_options(pmars-sdl3-events PRIVATE -mconsole -static -static-libgcc -static-libstdc++)
  endif()
endif()
